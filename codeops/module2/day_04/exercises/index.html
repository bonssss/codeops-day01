Key CSS Rules 
/* Multi-Column Desktop Grid */
.ex4-page-grid {
  display: grid;
  grid-template-areas:
    "header  header"
    "sidebar main"
    "footer  footer";
  grid-template-columns: 220px 1fr;
  grid-template-rows: auto 1fr auto;
}

/* Single-Column Collapse under 700px */
@media (max-width: 700px) {
  .ex4-page-grid {
    grid-template-areas:
      "header"
      "sidebar"
      "main"
      "footer";
    grid-template-columns: 1fr;
  }
}
